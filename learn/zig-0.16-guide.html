第 1 章 · 01_hello.zig

Hello World と標準出力

0.16 の入口は main(init: std.process.Init)。I/O に必要な io やアロケータはここから受け取ります。

  • init.io が Io 値。外界に触れる処理にはこれを渡す
  • Io.File.stdout().writeStreamingAll(io, ...) はバッファなしの最短ルート
  • 書式付き出力は Io.File.Writer にバッファを渡し、&writer.interface に print する
  • バッファ付き出力は最後に flush() を忘れない
  • 旧 std.io.getStdOut() は存在しない
実行zig run learn/01_hello.zig
ソースコード全文(01_hello.zig)
//! 第 1 章: Hello, World と標準出力
//!
//! 実行: zig run learn/01_hello.zig
//!
//! 0.16 のポイント
//!   - main は `std.process.Init` を受け取れる(通称 Juicy Main)
//!   - I/O は暗黙のグローバルではなく、`Io` 値を明示的に渡して行う
//!   - `init.io` が実行環境から渡される `Io`、`init.gpa` / `init.arena` がアロケータ
//!   - 0.15 以前の `std.io.getStdOut()` は存在しない

const std = @import("std");
const Io = std.Io;

pub fn main(init: std.process.Init) !void {
    const io = init.io;

    // 1. 一番簡単な方法: バッファなしで書き込む
    try Io.File.stdout().writeStreamingAll(io, "hello world!\n");

    // 2. 書式付き出力: バッファ付きの Writer を作る
    var buffer: [1024]u8 = undefined;
    var stdout_writer: Io.File.Writer = .init(.stdout(), io, &buffer);
    const out = &stdout_writer.interface;

    try out.print("{s} から {d} 回挨拶します\n", .{ "Zig 0.16", 3 });
    for (0..3) |i| {
        try out.print("  挨拶 {d}\n", .{i + 1});
    }

    // バッファに溜まった分は flush しないと出力されない(忘れがち)
    try out.flush();

    // 3. デバッグ用: stderr に書く。Io 不要で手軽
    std.debug.print("これは stderr です\n", .{});
}

第 2 章 · 02_basics.zig

変数・型・制御構文

各例は test ブロックで書かれています。値を書き換えてテストが失敗する様子も試してください。

  • 基本は const。再代入が要るときだけ var
  • 整数オーバーフローは Debug で panic。ラップさせるなら +%
  • 型変換は @intCast などで明示する
  • if / switch / ブロックは式として値を返せる
  • switch は網羅性が必須(else か全列挙)
実行zig test learn/02_basics.zig
ソースコード全文(02_basics.zig)
//! 第 2 章: 変数・型・制御構文
//!
//! 実行: zig test learn/02_basics.zig
//! このファイルは `test` ブロックで動作を確認します。
//! 値を変えて、テストが失敗するのを試してみましょう。

const std = @import("std");
const expect = std.testing.expect;
const expectEqual = std.testing.expectEqual;
const expectEqualStrings = std.testing.expectEqualStrings;

test "const と var" {
    const a = 10; // 変更不可。基本は const を使う
    var b: i32 = 20; // 変更可能。型を指定
    b += a;
    try expectEqual(30, b);

    // 未使用の変数・ローカル const はコンパイルエラーになる。
    // 意図的に使わないときは `_ = x;` と書く。
}

test "整数・浮動小数点・bool" {
    const x: u8 = 255;
    // オーバーフローは Debug では panic。ラップさせたいときは専用演算子 +% を使う
    try expectEqual(0, x +% 1);

    const y: f64 = 1.5;
    try expectEqual(3.0, y * 2);

    const flag: bool = x > 100 and y < 2;
    try expect(flag);

    // 型変換は明示的に。@as, @intCast, @floatFromInt などを使う
    const big: i64 = 1000;
    const small: i16 = @intCast(big);
    try expectEqual(1000, small);
}

test "配列とスライス" {
    const arr = [_]i32{ 1, 2, 3, 4, 5 }; // 長さはコンパイル時に決まる
    try expectEqual(5, arr.len);

    const slice: []const i32 = arr[1..4]; // 部分列(ポインタ + 長さ)
    try expectEqual(3, slice.len);
    try expectEqual(2, slice[0]);

    // 文字列は []const u8 (UTF-8 のバイト列)。"日本語" は 1 文字 3 バイト
    const s = "日本語";
    try expectEqual(9, s.len);
}

test "if / while / for" {
    // if は式。値を返せる
    const n = 7;
    const kind = if (n % 2 == 0) "偶数" else "奇数";
    try expectEqualStrings("奇数", kind);

    // while: 継続式 (: (i += 1)) を持てる
    var i: u32 = 0;
    var sum: u32 = 0;
    while (i < 5) : (i += 1) {
        sum += i;
    }
    try expectEqual(10, sum);

    // for: 要素と添字を同時に回す
    const items = [_]u32{ 10, 20, 30 };
    var total: u32 = 0;
    for (items, 0..) |item, idx| {
        total += item * @as(u32, @intCast(idx));
    }
    try expectEqual(80, total); // 0*10 + 1*20 + 2*30

    // 範囲のみ: 0..3 は 0,1,2
    var count: usize = 0;
    for (0..3) |_| count += 1;
    try expectEqual(3, count);
}

test "switch は網羅性が必須" {
    const score: u8 = 85;
    const grade = switch (score) {
        90...100 => 'A',
        80...89 => 'B',
        70...79 => 'C',
        else => 'D', // 全ての値を網羅しないとコンパイルエラー
    };
    try expectEqual('B', grade);
}

test "ラベル付きブロックと break" {
    const v = blk: {
        var acc: u32 = 1;
        for (1..5) |k| acc *= @intCast(k);
        break :blk acc;
    };
    try expectEqual(24, v);
}

fn add(a: i32, b: i32) i32 {
    return a + b;
}

test "関数" {
    try expectEqual(5, add(2, 3));
}

第 3 章 · 03_errors.zig

エラー処理・optional・defer

Zig に例外はありません。失敗は !T(error union)、不在は ?T(optional)という値で表します。

  • try はエラーなら呼び出し元へそのまま返す
  • catch で既定値に置換、または |err| で分岐
  • orelse で optional の既定値、.? は null なら panic
  • defer はスコープ終了時に逆順で実行
  • errdefer はエラーで抜けるときだけ実行(失敗時の後始末)
実行zig test learn/03_errors.zig
ソースコード全文(03_errors.zig)
//! 第 3 章: エラー処理・Optional・defer
//!
//! 実行: zig test learn/03_errors.zig
//!
//! Zig には例外がありません。失敗は「値」として扱います。
//!   - `!T`  : エラーまたは T (error union)
//!   - `?T`  : null または T (optional)

const std = @import("std");
const expect = std.testing.expect;
const expectEqual = std.testing.expectEqual;
const expectError = std.testing.expectError;

const DivError = error{DivideByZero};

fn divide(a: i32, b: i32) DivError!i32 {
    if (b == 0) return error.DivideByZero;
    return @divTrunc(a, b);
}

test "try / catch" {
    try expectEqual(3, try divide(9, 3));
    try expectError(error.DivideByZero, divide(1, 0));

    // catch でデフォルト値に置き換える
    const r = divide(1, 0) catch -1;
    try expectEqual(-1, r);

    // エラーごとに処理を分ける
    const msg = if (divide(1, 0)) |_| "ok" else |err| switch (err) {
        error.DivideByZero => "0 で割れません",
    };
    try std.testing.expectEqualStrings("0 で割れません", msg);
}

fn calc(a: i32, b: i32) !i32 {
    // try は「エラーならそのまま呼び出し元へ返す」の略記
    const q = try divide(a, b);
    return q + 1;
}

test "エラーの伝播" {
    try expectEqual(4, try calc(9, 3));
    try expectError(error.DivideByZero, calc(9, 0));
}

fn findIndex(items: []const i32, target: i32) ?usize {
    for (items, 0..) |item, i| {
        if (item == target) return i;
    }
    return null;
}

test "optional" {
    const items = [_]i32{ 5, 6, 7 };

    // if でアンラップ
    if (findIndex(&items, 6)) |idx| {
        try expectEqual(1, idx);
    } else {
        return error.TestUnexpectedResult;
    }

    // orelse でデフォルト値
    try expectEqual(0, findIndex(&items, 99) orelse 0);

    // .? は null なら panic(確実に非 null と分かっているときだけ)
    try expectEqual(2, findIndex(&items, 7).?);
}

test "defer と errdefer" {
    var log: [3]u8 = undefined;
    var n: usize = 0;

    {
        // defer はスコープを抜けるときに(逆順で)実行される
        defer {
            log[n] = 'C';
            n += 1;
        }
        defer {
            log[n] = 'B';
            n += 1;
        }
        log[n] = 'A';
        n += 1;
    }
    try std.testing.expectEqualStrings("ABC", &log);
}

var cleaned_up = false;

fn mayFail(fail: bool) !void {
    // errdefer はエラーで関数を抜けるときだけ実行される。
    // 確保したリソースを失敗時に解放する用途で使う。
    errdefer cleaned_up = true;
    if (fail) return error.Oops;
}

test "errdefer" {
    cleaned_up = false;
    try mayFail(false);
    try expect(!cleaned_up);

    try expectError(error.Oops, mayFail(true));
    try expect(cleaned_up);
}

第 4 章 · 04_memory.zig

アロケータとコレクション

グローバルなメモリ確保はありません。確保する関数はアロケータを引数に取り、解放責任は呼び出し側にあります。

  • テストでは std.testing.allocator がリークを自動検出
  • ArrayList は .empty で初期化し、append(gpa, x) と都度アロケータを渡す
  • ArenaAllocator は deinit で一括解放
  • FixedBufferAllocator はヒープを使わず、足りなければ OutOfMemory
  • アプリでは init.gpa / init.arena を使う
実行zig test learn/04_memory.zig
ソースコード全文(04_memory.zig)
//! 第 4 章: アロケータとコレクション
//!
//! 実行: zig test learn/04_memory.zig
//!
//! Zig にはグローバルなメモリ確保がありません。確保が必要な関数は
//! `std.mem.Allocator` を引数で受け取り、呼び出し側が解放の責任を持ちます。
//!
//! 0.16 のポイント
//!   - ArrayList は「アロケータを保持しない」形(`.empty` で初期化)が標準。
//!     append / deinit のたびにアロケータを渡す。
//!   - テストでは `std.testing.allocator` を使うとリークを自動検出してくれる。
//!   - アプリでは `init.gpa`(汎用)や `init.arena`(一括解放)が渡される。

const std = @import("std");
const testing = std.testing;
const expectEqual = testing.expectEqual;

test "alloc / free" {
    const gpa = testing.allocator;

    const buf = try gpa.alloc(u8, 4);
    defer gpa.free(buf); // コメントアウトするとリークとして検出される
    @memset(buf, 'z');
    try testing.expectEqualStrings("zzzz", buf);

    // 1 個だけ確保するなら create / destroy
    const p = try gpa.create(i32);
    defer gpa.destroy(p);
    p.* = 42;
    try expectEqual(42, p.*);
}

test "ArrayList" {
    const gpa = testing.allocator;

    var list: std.ArrayList(i32) = .empty;
    defer list.deinit(gpa);

    try list.append(gpa, 1);
    try list.append(gpa, 2);
    try list.appendSlice(gpa, &.{ 3, 4 });

    try expectEqual(4, list.items.len);
    try expectEqual(3, list.items[2]);
    try expectEqual(4, list.pop()); // 0.16 では ?T を返す
}

test "HashMap" {
    const gpa = testing.allocator;

    var map: std.StringHashMapUnmanaged(u32) = .empty;
    defer map.deinit(gpa);

    try map.put(gpa, "apple", 3);
    try map.put(gpa, "banana", 5);

    try expectEqual(3, map.get("apple").?);
    try expectEqual(null, map.get("cherry"));
    try expectEqual(2, map.count());
}

test "ArenaAllocator は一括解放" {
    // arena は個別の free が不要。deinit で全てまとめて解放する。
    // リクエスト単位・コマンド単位の処理など、寿命が揃うデータに向く。
    var arena: std.heap.ArenaAllocator = .init(testing.allocator);
    defer arena.deinit();
    const a = arena.allocator();

    const x = try a.alloc(u8, 100);
    const y = try a.dupe(u8, "hello");
    _ = x;
    try testing.expectEqualStrings("hello", y);
    // free を呼ばなくてもリークにならない
}

test "FixedBufferAllocator はスタック上で完結" {
    var storage: [16]u8 = undefined;
    var fba: std.heap.FixedBufferAllocator = .init(&storage);
    const a = fba.allocator();

    _ = try a.alloc(u8, 10);
    try testing.expectError(error.OutOfMemory, a.alloc(u8, 10));
}

test "std.fmt で文字列を作る" {
    const gpa = testing.allocator;
    const s = try std.fmt.allocPrint(gpa, "{s}-{d}", .{ "id", 7 });
    defer gpa.free(s);
    try testing.expectEqualStrings("id-7", s);

    // バッファを渡す版ならアロケータ不要
    var buf: [16]u8 = undefined;
    const t = try std.fmt.bufPrint(&buf, "{x}", .{255});
    try testing.expectEqualStrings("ff", t);
}

第 5 章 · 05_types.zig

構造体・enum・union・comptime

ジェネリクスは「型を受け取って型を返す関数」で表現します。特別な構文はありません。

  • struct 内の関数はメソッド。self を取れば p.length() と呼べる
  • union(enum) は switch で安全に取り出す
  • fn Pair(comptime T: type) type がジェネリック型の基本形
  • comptime を付けるとコンパイル時に評価
  • 0.16 では @Type が廃止され @Int などに置き換え
実行zig test learn/05_types.zig
ソースコード全文(05_types.zig)
//! 第 5 章: 構造体・enum・union・comptime
//!
//! 実行: zig test learn/05_types.zig

const std = @import("std");
const testing = std.testing;
const expectEqual = testing.expectEqual;

// ---- struct ----
const Point = struct {
    x: f32,
    y: f32 = 0, // デフォルト値

    // 構造体の中の関数 = メソッド。第 1 引数が self なら `p.length()` で呼べる
    fn length(self: Point) f32 {
        return @sqrt(self.x * self.x + self.y * self.y);
    }

    fn scale(self: *Point, k: f32) void {
        self.x *= k;
        self.y *= k;
    }
};

test "struct" {
    var p: Point = .{ .x = 3, .y = 4 }; // 型は左辺から推論される
    try expectEqual(5, p.length());
    p.scale(2);
    try expectEqual(10, p.length());

    const q: Point = .{ .x = 1 };
    try expectEqual(0, q.y);
}

// ---- enum ----
const Color = enum {
    red,
    green,
    blue,

    fn isWarm(self: Color) bool {
        return self == .red;
    }
};

test "enum" {
    const c: Color = .red;
    try testing.expect(c.isWarm());
    try testing.expectEqualStrings("blue", @tagName(Color.blue));
}

// ---- tagged union ----
const Shape = union(enum) {
    circle: f32, // 半径
    rect: struct { w: f32, h: f32 },

    fn area(self: Shape) f32 {
        return switch (self) {
            .circle => |r| std.math.pi * r * r,
            .rect => |r| r.w * r.h,
        };
    }
};

test "tagged union" {
    const s: Shape = .{ .rect = .{ .w = 2, .h = 3 } };
    try expectEqual(6, s.area());
    const c: Shape = .{ .circle = 1 };
    try testing.expectApproxEqAbs(3.14159, c.area(), 0.001);
}

// ---- comptime とジェネリクス ----
// Zig のジェネリクスは「型を受け取って型を返す関数」で表現する
fn Pair(comptime T: type) type {
    return struct {
        a: T,
        b: T,

        const Self = @This();

        fn swap(self: Self) Self {
            return .{ .a = self.b, .b = self.a };
        }
    };
}

test "ジェネリックな型" {
    const p: Pair(i32) = .{ .a = 1, .b = 2 };
    try expectEqual(2, p.swap().a);

    const s: Pair([]const u8) = .{ .a = "x", .b = "y" };
    try testing.expectEqualStrings("y", s.swap().a);
}

// anytype: 引数の型を呼び出し側から受け取る(コンパイル時に具体化される)
fn double(x: anytype) @TypeOf(x) {
    return x * 2;
}

test "anytype" {
    try expectEqual(4, double(@as(i32, 2)));
    try expectEqual(5.0, double(@as(f64, 2.5)));
}

// comptime: コンパイル時に実行される計算
fn fib(n: u32) u32 {
    return if (n < 2) n else fib(n - 1) + fib(n - 2);
}

test "comptime" {
    const v = comptime fib(10); // コンパイル時に計算済み
    try expectEqual(55, v);

    // 0.16 では @Type が廃止され、@Int / @Struct などの専用組み込み関数になった
    const U12 = @Int(.unsigned, 12);
    try expectEqual(12, @bitSizeOf(U12));
}

第 6 章 · 06_file_io.zig

ファイル I/O と引数

0.16 最大の変更点。ファイルや環境など外界に触れる操作は、すべて io を受け取ります。

  • std.fs.cwd() → Io.Dir.cwd()
  • cwd.writeFile(io, .{ .sub_path, .data }) で書き込み
  • cwd.readFileAlloc(io, path, gpa, .limited(n)) で読み込み。上限超過は StreamTooLong
  • 引数は init.minimal.args.toSlice(...)、環境変数は init.environ_map
  • openFile の失敗は error.FileNotFound などで分岐できる
実行zig run learn/06_file_io.zig -- foo bar
ソースコード全文(06_file_io.zig)
//! 第 6 章: ファイル I/O とコマンドライン引数
//!
//! 実行: zig run learn/06_file_io.zig -- foo bar
//!
//! 0.16 の最大の変更点。ファイル、時計、プロセスなど外界に触れる操作は
//! すべて `Io` 値を受け取ります。旧 `std.fs.cwd()` は `std.Io.Dir.cwd()` に、
//! ファイル操作のメソッドは第 1 引数に `io` を取る形に変わりました。

const std = @import("std");
const Io = std.Io;

pub fn main(init: std.process.Init) !void {
    const io = init.io;
    const gpa = init.gpa;

    var stdout_buf: [1024]u8 = undefined;
    var stdout_writer: Io.File.Writer = .init(.stdout(), io, &stdout_buf);
    const out = &stdout_writer.interface;

    // ---- コマンドライン引数 ----
    // arena は main の終わりまで生きる。ここで確保したものは解放不要
    const args = try init.minimal.args.toSlice(init.arena.allocator());
    for (args, 0..) |arg, i| {
        try out.print("args[{d}] = {s}\n", .{ i, arg });
    }

    // ---- 環境変数 ----
    // 0.16 では環境変数もグローバル関数ではなく init 経由
    if (init.environ_map.get("HOME")) |home| {
        try out.print("HOME = {s}\n", .{home});
    }

    // ---- 書き込み ----
    const cwd = Io.Dir.cwd();
    const path = ".zig-cache/learn_06_sample.txt";

    try cwd.writeFile(io, .{
        .sub_path = path,
        .data = "1 行目\n2 行目\n3 行目\n",
    });

    // ---- まとめて読み込み ----
    // 第 5 引数は読み込む上限サイズ。超えると error.StreamTooLong
    const content = try cwd.readFileAlloc(io, path, gpa, .limited(1024));
    defer gpa.free(content);
    try out.print("読み込んだ: {d} バイト\n", .{content.len});

    // ---- 行ごとに処理 ----
    var lines = std.mem.splitScalar(u8, content, '\n');
    var n: usize = 0;
    while (lines.next()) |line| {
        if (line.len == 0) continue;
        n += 1;
        try out.print("  {d}: {s}\n", .{ n, line });
    }

    // ---- 存在しないファイルのエラー処理 ----
    if (cwd.openFile(io, "no_such_file.txt", .{})) |file| {
        file.close(io);
    } else |err| switch (err) {
        error.FileNotFound => try out.print("no_such_file.txt は無い\n", .{}),
        else => return err,
    }

    try cwd.deleteFile(io, path);
    try out.flush();
}

第 7 章 · 07_concurrency.zig

非同期処理と並行処理

Io が async を提供します。どの実装(スレッドプール等)で動くかは呼び出し側が決め、関数側は io を受け取るだけです。

  • io.async(f, .{args}) と future.await(io)
  • io.concurrent は本当に並行実行することを要求
  • Io.Group で複数タスクをまとめて待つ。defer group.cancel(io) で後始末
  • io.sleep(.fromMilliseconds(n), .awake) で待機
  • この例では 0.1 秒の処理 2 つが合計約 105ms で終わる
実行zig run learn/07_concurrency.zig
ソースコード全文(07_concurrency.zig)
//! 第 7 章: 非同期処理と並行処理(std.Io の本領)
//!
//! 実行: zig run learn/07_concurrency.zig
//!
//! 0.16 では「関数に色が付かない」async を `Io` が提供します。
//!   - `io.async(f, .{args})`     : f を非同期に呼ぶ。await で結果を受け取る。
//!                                  Io の実装次第で、その場で逐次実行されることもある。
//!   - `io.concurrent(f, .{args})`: 本当に並行実行することを要求する。
//!   - `Io.Group`                 : 複数のタスクをまとめて待つ。
//! どの実装(スレッドプール等)で動かすかは main を呼ぶ側が決めており、
//! 呼び出されるコードは `io` を受け取るだけで済みます。

const std = @import("std");
const Io = std.Io;

fn slowAdd(io: Io, a: i32, b: i32) i32 {
    io.sleep(.fromMilliseconds(100), .awake) catch {};
    return a + b;
}

fn worker(io: Io, name: []const u8, ms: i64) void {
    io.sleep(.fromMilliseconds(ms), .awake) catch return;
    std.debug.print("  {s} 完了 ({d}ms)\n", .{ name, ms });
}

pub fn main(init: std.process.Init) !void {
    const io = init.io;

    // ---- async / await ----
    std.debug.print("[async / await]\n", .{});
    const t0 = Io.Clock.awake.now(io);

    var f1 = io.async(slowAdd, .{ io, 1, 2 });
    var f2 = io.async(slowAdd, .{ io, 10, 20 });
    const r1 = f1.await(io);
    const r2 = f2.await(io);

    const elapsed = t0.untilNow(io, .awake);
    std.debug.print("  結果: {d}, {d} / 経過 {d}ms\n", .{
        r1, r2, elapsed.toMilliseconds(),
    });

    // ---- Group: 複数タスクをまとめて待つ ----
    std.debug.print("[Group]\n", .{});
    var group: Io.Group = .init;
    defer group.cancel(io); // 途中でエラー終了しても確実に後始末する
    group.async(io, worker, .{ io, "A", 300 });
    group.async(io, worker, .{ io, "B", 100 });
    group.async(io, worker, .{ io, "C", 200 });
    try group.await(io);
    std.debug.print("  全タスク完了\n", .{});
}

第 8 章 · 08_pointers.zig

ポインタ

Zig のポインタは種類ごとに型が分かれています。*T(単一)、[*]T(長さ不明)、[]T(スライス)、?*T(null 可)。

  • &x でアドレス、p.* で参照先
  • 引数は常に値渡し。変更させたいときはポインタを渡す
  • for (buf) |*b| で要素のポインタを受け取り書き換える
  • *T は null になれず、null を許すなら ?*T
  • 連結リストなど、ポインタの実用例も収録
実行zig test learn/08_pointers.zig
ソースコード全文(08_pointers.zig)
//! 第 8 章(基本文法編): ポインタ
//!
//! 実行: zig test learn/08_pointers.zig
//!
//! Zig のポインタは種類ごとに型が分かれています。
//!   *T         単一要素へのポインタ(演算不可)
//!   [*]T       複数要素へのポインタ(長さ不明、Cの配列ポインタ相当)
//!   []T        スライス(ポインタ + 長さ)。普段はこれを使う
//!   *const T   読み取り専用
//!   ?*T        null になりうるポインタ

const std = @import("std");
const testing = std.testing;
const expectEqual = testing.expectEqual;

test "単一要素ポインタ" {
    var x: i32 = 10;
    const p: *i32 = &x; // & でアドレスを取る
    p.* = 20; // .* で参照先へアクセス
    try expectEqual(20, x);
}

fn increment(n: *u32) void {
    n.* += 1;
}

test "参照渡し" {
    // 引数は常に値渡し。変更させたいときはポインタを渡す
    var count: u32 = 0;
    increment(&count);
    increment(&count);
    try expectEqual(2, count);
}

test "const ポインタ" {
    const x: i32 = 5;
    const p: *const i32 = &x;
    try expectEqual(5, p.*);
    // p.* = 6; // コンパイルエラー: const への書き込み
}

test "配列へのポインタとスライス" {
    var arr = [_]u8{ 1, 2, 3, 4 };

    // 配列へのポインタ(*[4]u8)はスライスに自動で変換される
    const slice: []u8 = &arr;
    slice[0] = 100;
    try expectEqual(100, arr[0]);

    // スライスの一部を取る
    const mid = arr[1..3];
    try expectEqual(2, mid.len);
    try expectEqual(@as(usize, 4), arr.len);

    // スライスの要素のポインタ
    const first: *u8 = &slice[0];
    first.* = 7;
    try expectEqual(7, arr[0]);
}

fn fill(buf: []u8, value: u8) void {
    // スライスは参照の一種。要素の書き換えは呼び出し元にも見える
    for (buf) |*b| b.* = value; // |*b| で要素のポインタを受け取る
}

test "for で要素を書き換える" {
    var arr = [_]u8{ 0, 0, 0 };
    fill(&arr, 9);
    try testing.expectEqualSlices(u8, &.{ 9, 9, 9 }, &arr);
}

test "optional ポインタ" {
    var value: i32 = 1;
    var maybe: ?*i32 = null;
    try expectEqual(null, maybe);

    maybe = &value;
    if (maybe) |p| p.* += 1;
    try expectEqual(2, value);
    // ?*T は null を表現でき、C のように「うっかり null」にならない。
    // *T は null になれない。
}

const Node = struct {
    value: i32,
    next: ?*Node = null,
};

test "連結リスト(ポインタの実用例)" {
    var c: Node = .{ .value = 3 };
    var b: Node = .{ .value = 2, .next = &c };
    var a: Node = .{ .value = 1, .next = &b };
    _ = .{ &a, &b, &c };

    var sum: i32 = 0;
    var cur: ?*Node = &a;
    while (cur) |node| : (cur = node.next) {
        sum += node.value;
    }
    try expectEqual(6, sum);
}

test "ポインタキャストは明示的に" {
    var x: u32 = 0x01020304;
    const bytes: *[4]u8 = @ptrCast(&x);
    // ネイティブのエンディアンに依存する。確認するだけで値は固定しない
    try expectEqual(4, bytes.len);
}

第 9 章 · 09_strings.zig

文字列

Zig に文字列型はなく、UTF-8 のバイト列 []const u8 で表します。len はバイト数です。

  • 比較は == ではなく std.mem.eql
  • 検索・分割・トリムは std.mem の関数
  • コンパイル時は ++、実行時の連結はアロケータ付きの mem.concat / mem.join
  • 数値変換は std.fmt.parseInt / parseFloat
  • 複数行リテラルは \\ で始める

//! 第 9 章(基本文法編): 文字列
//!
//! 実行: zig test learn/09_strings.zig
//!
//! Zig に文字列型はなく、文字列は UTF-8 のバイト列 `[]const u8` です。
//! リテラル "abc" の型は `*const [3:0]u8`(末尾に 0 が付く配列へのポインタ)で、

const expectEqual = testing.expectEqual;

test "文字列はバイト列" {

test "比較は == ではなく mem.eql" {
    const a = "hello";
    const b: []const u8 = "hello";
    // `a == b` はスライス同士なのでコンパイルエラー。中身の比較は mem.eql
    try testing.expect(mem.eql(u8, a, b));
    try expectEqualStrings("hello", b); // テストでは expectEqualStrings が便利
}

test "検索・判定" {
    const s = "hello, world";
    try expectEqual(7, mem.indexOf(u8, s, "world").?);
    try expectEqual(4, mem.indexOfScalar(u8, s, 'o').?);
    try testing.expect(mem.startsWith(u8, s, "hello"));
    try testing.expect(mem.endsWith(u8, s, "world"));
    try testing.expect(mem.containsAtLeast(u8, s, 2, "o"));
}

test "分割・トリム" {
    var it = mem.splitScalar(u8, "a,b,c", ',');
    try expectEqualStrings("a", it.next().?);
    try expectEqualStrings("b", it.next().?);
    try expectEqualStrings("c", it.next().?);
    try expectEqual(null, it.next());

    // 連続する区切りを 1 つとして扱う場合は tokenize
    var tok = mem.tokenizeScalar(u8, "  a   b  ", ' ');
    try expectEqualStrings("a", tok.next().?);
    try expectEqualStrings("b", tok.next().?);

    try expectEqualStrings("x", mem.trim(u8, "  x\n", " \n"));
}

test "連結" {
    // コンパイル時なら ++ で連結できる
    const s = "foo" ++ "bar";
    try expectEqualStrings("foobar", s);

    // 実行時の連結はアロケータが必要
    const gpa = testing.allocator;
    const joined = try mem.concat(gpa, u8, &.{ "ab", "cd", "ef" });
    defer gpa.free(joined);
    try expectEqualStrings("abcdef", joined);

    const csv = try mem.join(gpa, ",", &.{ "x", "y", "z" });
    defer gpa.free(csv);
    try expectEqualStrings("x,y,z", csv);
}

test "文字の種類" {
    try testing.expect(std.ascii.isDigit('7'));
    try testing.expect(std.ascii.isAlphabetic('a'));
    try expectEqual('A', std.ascii.toUpper('a'));

    var buf: [5]u8 = undefined;
    const upper = std.ascii.upperString(&buf, "hello");
    try expectEqualStrings("HELLO", upper);
}

test "数値との変換" {
    try expectEqual(123, try std.fmt.parseInt(i32, "123", 10));
    try expectEqual(255, try std.fmt.parseInt(u8, "ff", 16));
    try testing.expectError(error.InvalidCharacter, std.fmt.parseInt(i32, "12a", 10));
    try expectEqual(1.5, try std.fmt.parseFloat(f64, "1.5"));
}

test "UTF-8 を文字(コードポイント)単位で扱う" {
    const view = try std.unicode.Utf8View.init("日本語");
    var it = view.iterator();
    var n: usize = 0;
    while (it.nextCodepoint()) |_| n += 1;
    try expectEqual(3, n);
}

test "複数行文字列リテラル" {
    const text =
        \\1 行目
        \\2 行目
    ;
    try expectEqualStrings("1 行目\n2 行目", text);
}

test "書式指定" {
    var buf: [64]u8 = undefined;
    const s = try std.fmt.bufPrint(&buf, "{s}|{d:>5}|{x}|{b}|{d:.2}", .{ "id", 42, 255, 5, 3.14159 });
    try expectEqualStrings("id|   42|ff|101|3.14", s);
}

const Stack = struct {
    items: [4]i32 = undefined,
    len: usize = 0,

    fn push(self: *Stack, v: i32) !void {
        if (self.len == self.items.len) return error.Full;
        self.items[self.len] = v;
        self.len += 1;
    }

    fn pop(self: *Stack) ?i32 {
        if (self.len == 0) return null;
        self.len -= 1;
        return self.items[self.len];
    }
};

    try s.push(2);
    try testing.expectEqual(2, s.pop());

    try testing.expectEqual(null, s.pop());
}

第 12 章 · 12_control_flow.zig

制御構文の発展

Zig は「値を取り出しながら分岐する」キャプチャ構文が一貫しています。

  • inline for はコンパイル時に展開される
  • ラベル付き break :outer / continue :outer
  • for ... else は式として値を返せる

    for (tokens) |t| {

    try expectEqual(140, dot);
}

test "ラベル付き break / continue" {
    var found: ?struct { usize, usize } = null;
    outer: for (0..5) |i| {
        for (0..5) |j| {
            if (i * j == 6) {
                found = .{ i, j };
                break :outer; // 外側のループまで抜ける
            }
            if (j > i) continue :outer;
        }
    }
    try expectEqual(2, found.?[0]);
    try expectEqual(3, found.?[1]);
}

test "else 付きの for / while は式として値を返せる" {
    const items = [_]i32{ 1, 3, 5 };
    const first_even: ?i32 = for (items) |v| {
        if (@mod(v, 2) == 0) break v;
    } else null;
    try expectEqual(null, first_even);
}
